using System;

class Libro
{
    public string titulo;
    public string autor;
    public int anioPublicacion;
    public bool disponible;

    public Libro(string t, string a, int anio, bool disp)
    {
        titulo = t;
        autor = a;
        anioPublicacion = anio;
        disponible = disp;
    }

    public void mostrarInformacion()
    {
        Console.WriteLine("Título: " + titulo + ", Autor: " + autor + ", Año: " + anioPublicacion + ", Disponible: " + disponible);
    }

    public void prestarLibro()
    {
        if (disponible == true)
        {
            disponible = false;
            Console.WriteLine("El libro '" + titulo + "' ha sido prestado.");
        }
        else
        {
            Console.WriteLine("El libro '" + titulo + "' no está disponible.");
        }
    }

    public void devolverLibro()
    {
        disponible = true;
        Console.WriteLine("El libro '" + titulo + "' ha sido devuelto.");
    }
}

class Mascota
{
    public string nombre;
    public string especie;
    public int edad;
    public bool vacunado;

    public Mascota(string n, string e, int ed, bool vac)
    {
        nombre = n;
        especie = e;
        edad = ed;
        vacunado = vac;
    }

    public void mostrarInformacion()
    {
        Console.WriteLine("Nombre: " + nombre + ", Especie: " + especie + ", Edad: " + edad + " años, Vacunado: " + vacunado);
    }

    public void vacunar()
    {
        vacunado = true;
        Console.WriteLine(nombre + " ha sido vacunado.");
    }

    public void cumplirAnios()
    {
        edad = edad + 1;
        Console.WriteLine(nombre + " cumplió años. Ahora tiene " + edad + " años.");
    }
}

class Estudiante
{
    public string nombre;
    public int edad;
    public string grado;
    public double[] notas;

    public Estudiante(string n, int e, string g, double[] nts)
    {
        nombre = n;
        edad = e;
        grado = g;
        notas = nts;
    }

    public double calcularPromedio()
    {
        double suma = 0;
        for (int i = 0; i < notas.Length; i++)
        {
            suma = suma + notas[i];
        }
        return suma / notas.Length;
    }

    public void mostrarInformacion()
    {
        Console.WriteLine("Nombre: " + nombre + ", Edad: " + edad + ", Grado: " + grado + ", Promedio: " + calcularPromedio());
    }

    public void aprobar()
    {
        if (calcularPromedio() >= 65)
        {
            Console.WriteLine("El estudiante " + nombre + " aprobó.");
        }
        else
        {
            Console.WriteLine("El estudiante " + nombre + " reprobó.");
        }
    }

    public void agregarNota(double nuevaNota)
    {
        double[] nuevasNotas = new double[notas.Length + 1];

        for (int i = 0; i < notas.Length; i++)
        {
            nuevasNotas[i] = notas[i];
        }

        nuevasNotas[notas.Length] = nuevaNota;

        notas = nuevasNotas;
        Console.WriteLine("Nota " + nuevaNota + " agregada correctamente a " + nombre + ".");
    }
}
