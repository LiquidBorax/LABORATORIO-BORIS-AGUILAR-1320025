using System;

class Program
{
    static void Main()
    {

        int opcion = 0;

        do
        {
            Console.WriteLine("\n--- MENU ---");
            Console.WriteLine("1. Suma de filas y columnas");
            Console.WriteLine("2. Numero mayor en matriz");
            Console.WriteLine("3. Multiplicacion de matrices");
            Console.WriteLine("4. Diagonales de matriz");
            Console.WriteLine("5. Salir");
            Console.Write("Seleccione una opcion: ");

            opcion = Convert.ToInt32(Console.ReadLine());

            switch (opcion)
            {
                case 1:
                    ejercicio1();
                    break;

                case 2:
                    ejercicio2();
                    break;

                case 3:
                    ejercicio3();
                    break;

                case 4:
                    ejercicio4();
                    break;
            }

        } while (opcion != 5);
    }

    // ---------------- EJERCICIO 1 ----------------

    static void llenarMatriz1(int[,] m)
    {
        for (int i = 0; i < 4; i++)
        {
            for (int j = 0; j < 4; j++)
            {
                Console.Write("Ingrese dato [" + i + "," + j + "]: ");
                m[i, j] = Convert.ToInt32(Console.ReadLine());
            }
        }
    }

    static int sumaFila(int[,] m, int fila)
    {
        int suma = 0;

        for (int j = 0; j < 4; j++)
        {
            suma = suma + m[fila, j];
        }

        return suma;
    }

    static int sumaColumna(int[,] m, int col)
    {
        int suma = 0;

        for (int i = 0; i < 4; i++)
        {
            suma = suma + m[i, col];
        }

        return suma;
    }

    static void ejercicio1()
    {
        int[,] matriz = new int[4, 4];

        llenarMatriz1(matriz);

        Console.Write("Ingrese fila a sumar (0-3): ");
        int fila = Convert.ToInt32(Console.ReadLine());

        Console.Write("Ingrese columna a sumar (0-3): ");
        int col = Convert.ToInt32(Console.ReadLine());

        Console.WriteLine("Suma fila: " + sumaFila(matriz, fila));
        Console.WriteLine("Suma columna: " + sumaColumna(matriz, col));
    }

    // ---------------- EJERCICIO 2 ----------------

    static void cargarMatriz(float[,] m)
    {
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 5; j++)
            {
                Console.Write("Ingrese dato [" + i + "," + j + "]: ");
                m[i, j] = Convert.ToSingle(Console.ReadLine());
            }
        }
    }

    static float mayorMatriz(float[,] m)
    {
        float mayor = m[0, 0];

        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 5; j++)
            {
                if (m[i, j] > mayor)
                {
                    mayor = m[i, j];
                }
            }
        }

        return mayor;
    }

    static void ejercicio2()
    {
        float[,] matriz = new float[3, 5];

        cargarMatriz(matriz);

        Console.WriteLine("Numero mayor: " + mayorMatriz(matriz));
    }

    // ---------------- EJERCICIO 3 ----------------

    static void llenarMatrizA(int[,] m)
    {
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 2; j++)
            {
                Console.Write("A[" + i + "," + j + "]: ");
                m[i, j] = Convert.ToInt32(Console.ReadLine());
            }
        }
    }

    static void llenarMatrizB(int[,] m)
    {
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 2; j++)
            {
                Console.Write("B[" + i + "," + j + "]: ");
                m[i, j] = Convert.ToInt32(Console.ReadLine());
            }
        }
    }

    static void multiplicar(int[,] A, int[,] B, int[,] R)
    {
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 2; j++)
            {
                R[i, j] = A[i, j] * B[i, j];
            }
        }
    }

    static void mostrarMatriz(int[,] m)
    {
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 2; j++)
            {
                Console.Write(m[i, j] + " ");
            }

            Console.WriteLine();
        }
    }

    static void ejercicio3()
    {
        int[,] A = new int[3, 2];
        int[,] B = new int[3, 2];
        int[,] R = new int[3, 2];

        llenarMatrizA(A);
        llenarMatrizB(B);

        multiplicar(A, B, R);

        Console.WriteLine("Matriz resultante:");

        mostrarMatriz(R);
    }

    // ---------------- EJERCICIO 4 ----------------

    static void llenar(int[,] m)
    {
        for (int i = 0; i < 5; i++)
        {
            for (int j = 0; j < 5; j++)
            {
                Console.Write("Ingrese dato [" + i + "," + j + "]: ");
                m[i, j] = Convert.ToInt32(Console.ReadLine());
            }
        }
    }

    static int sumaDiagonalPrincipal(int[,] m)
    {
        int suma = 0;

        for (int i = 0; i < 5; i++)
        {
            suma = suma + m[i, i];
        }

        return suma;
    }

    static int sumaDiagonalSecundaria(int[,] m)
    {
        int suma = 0;

        for (int i = 0; i < 5; i++)
        {
            suma = suma + m[i, 4 - i];
        }

        return suma;
    }

    static void ejercicio4()
    {
        int[,] matriz = new int[5, 5];

        llenar(matriz);

        Console.WriteLine("Suma diagonal principal: " + sumaDiagonalPrincipal(matriz));
        Console.WriteLine("Suma diagonal secundaria: " + sumaDiagonalSecundaria(matriz));
    }
}
